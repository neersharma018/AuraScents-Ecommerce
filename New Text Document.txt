/* ============================================
   PARTICLES (CANVAS)
============================================ */
function initParticles(canvasId, count) {
  var canvas = document.getElementById(canvasId);
  if (!canvas) return;
  var ctx = canvas.getContext('2d');
  var particles = [];
  var dpr = window.devicePixelRatio || 1;
  var cssW = 0, cssH = 0;
  
  function resize() {
    var rect = canvas.getBoundingClientRect();
    cssW = rect.width;
    cssH = rect.height;
    canvas.width = cssW * dpr;
    canvas.height = cssH * dpr;
    canvas.style.width = cssW + 'px';
    canvas.style.height = cssH + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  }
  resize();
  window.addEventListener('resize', resize);
  
  function createParticle() {
    return {
      x: Math.random() * cssW,
      y: Math.random() * cssH,
      r: Math.random() * 2 + 0.5,
      vy: -Math.random() * 0.4 - 0.1,
      vx: (Math.random() - 0.5) * 0.2,
      opacity: Math.random() * 0.6 + 0.2,
      twinkleSpeed: Math.random() * 0.02 + 0.005,
      twinklePhase: Math.random() * Math.PI * 2
    };
  }
  
  for (var i = 0; i < count; i++) particles.push(createParticle());
  
  function animate() {
    ctx.clearRect(0, 0, cssW, cssH);
    
    for (var i = 0; i < particles.length; i++) {
      var p = particles[i];
      p.y += p.vy;
      p.x += p.vx;
      p.twinklePhase += p.twinkleSpeed;
      
      var opacity = p.opacity * (0.5 + 0.5 * Math.sin(p.twinklePhase));
      
      var gradient = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.r * 4);
      gradient.addColorStop(0, 'rgba(244, 208, 63, ' + opacity + ')');
      gradient.addColorStop(0.4, 'rgba(212, 175, 55, ' + (opacity * 0.4) + ')');
      gradient.addColorStop(1, 'rgba(212, 175, 55, 0)');
      ctx.fillStyle = gradient;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.r * 4, 0, Math.PI * 2);
      ctx.fill();
      
      ctx.fillStyle = 'rgba(248, 244, 238, ' + opacity + ')';
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
      ctx.fill();
      
      if (p.y < -10 || p.x < -10 || p.x > cssW + 10) {
        particles[i] = createParticle();
        particles[i].y = cssH + 10;
      }
    }
    
    requestAnimationFrame(animate);
  }
  animate();
}

/* ============================================
   SCROLL PROGRESS & NAVBAR
============================================ */
function initScroll() {
  var progress = document.getElementById('scrollProgress');
  var navbar = document.getElementById('navbar');
  var navLinks = document.querySelectorAll('.nav-link');
  var sections = ['home', 'collections', 'bestsellers', 'about', 'journal', 'contact'];
  
  window.addEventListener('scroll', function() {
    var scrolled = window.scrollY;
    var total = document.body.scrollHeight - window.innerHeight;
    var percent = (scrolled / total) * 100;
    progress.style.width = percent + '%';
    
    if (scrolled > 50) navbar.classList.add('scrolled');
    else navbar.classList.remove('scrolled');
    
    var current = 'home';
    sections.forEach(function(id) {
      var el = document.getElementById(id);
      if (el && el.offsetTop - 120 <= scrolled) current = id;
    });
    navLinks.forEach(function(link) {
      link.classList.toggle('active', link.getAttribute('href') === '#' + current);
    });
  });
}

    el.addEventListener('mousemove', function(e) {
      var rect = el.getBoundingClientRect();
      var x = e.clientX - rect.left - rect.width / 2;
      var y = e.clientY - rect.top - rect.height / 2;
      el.style.transform = 'translate(' + (x * 0.25) + 'px, ' + (y * 0.35) + 'px)';

  });

  var observer = new IntersectionObserver(function(entries) {
    entries.forEach(function(entry) {
      if (entry.isIntersecting) {
        entry.target.classList.add('visible');
        observer.unobserve(entry.target);
      }
    });
  }, { threshold: 0.12, rootMargin: '0px 0px -50px 0px' });
  
  document.querySelectorAll('.reveal, .reveal-stagger').forEach(function(el) { observer.observe(el); });
}

/* ============================================
   HERO PARALLAX
============================================ */
function initParallax() {
  var bottle = document.getElementById('heroBottleWrap');
  var heroContent = document.getElementById('heroContent');
  if (!bottle) return;
  
  document.addEventListener('mousemove', function(e) {
    var x = (e.clientX / window.innerWidth - 0.5) * 20;
    var y = (e.clientY / window.innerHeight - 0.5) * 20;
    bottle.style.transform = 'translate(' + (x * 0.5) + 'px, ' + (y * 0.5) + 'px)';
    if (heroContent) heroContent.style.transform = 'translate(' + (-x * 0.2) + 'px, ' + (-y * 0.2) + 'px)';
  });
}

/* ============================================
   PRODUCT CAROUSEL
============================================ */
function initCarousel() {
  var track = document.getElementById('productsTrack');
  var prev = document.getElementById('prevProduct');
  var next = document.getElementById('nextProduct');
  if (!track || !prev || !next) return;
  
  var firstCard = track.querySelector('article');
  var scrollAmount = firstCard ? firstCard.offsetWidth + 24 : 340;
  
  prev.addEventListener('click', function() { track.scrollBy({ left: -scrollAmount, behavior: 'smooth' }); });
  next.addEventListener('click', function() { track.scrollBy({ left: scrollAmount, behavior: 'smooth' }); });
}

function initTestimonials() {
  var testimonials = [
    { text: 'AuraScents became my signature fragrance. The longevity is remarkable — I catch whispers of it on my scarf hours after application.', author: 'Isabelle Laurent', title: 'Fashion Editor, Paris' },
    { text: 'Beautiful packaging and amazing longevity. The Velvet Amber is unlike anything I have experienced — warm, intimate, and utterly addictive.', author: 'Marcus Chen', title: 'Collector, New York' },
    { text: 'The most luxurious perfume I have ever owned. Royal Oud commands attention without ever raising its voice. Pure sophistication.', author: 'Sophia Kovac', title: 'Art Director, Milan' }
  ];
  
  var wrap = document.getElementById('testimonialWrap');
  var dotsWrap = document.getElementById('testimonialDots');
  if (!wrap || !dotsWrap) return;
  
  wrap.innerHTML = testimonials.map(function(t, i) {
    return '<div class="testimonial-card glass-card p-10 md:p-14 text-center ' + (i === 0 ? 'active' : '') + '" data-index="' + i + '">' +
      '<div class="text-5xl serif text-gold-gradient mb-4" style="line-height: 1; font-weight: 300;">"</div>' +
      '<div class="stars mb-6">★★★★★</div>' +
      '<p class="serif text-2xl md:text-3xl mb-8 italic" style="line-height: 1.4; font-weight: 300;">' + t.text + '</p>' +
      '<div class="divider-gold mx-auto mb-4"></div>' +
      '<div class="text-sm tracking-[0.15em] uppercase" style="color: var(--gold);">' + t.author + '</div>' +
      '<div class="text-xs mt-1" style="color: var(--ivory-dim);">' + t.title + '</div>' +
    '</div>';
  }).join('');
  
  dotsWrap.innerHTML = testimonials.map(function(_, i) {
    return '<button class="w-8 h-1 transition-all ' + (i === 0 ? 'bg-[var(--gold)]' : 'bg-[var(--border-gold)]') + '" data-dot="' + i + '" aria-label="Testimonial ' + (i + 1) + '" style="border: none; cursor: pointer; padding: 0;"></button>';
  }).join('');
  
  var current = 0;
  var cards = wrap.querySelectorAll('.testimonial-card');
  var dots = dotsWrap.querySelectorAll('button');
  
  function goTo(idx) {
    cards[current].classList.remove('active');
    cards[current].classList.add('exit');
    dots[current].classList.remove('bg-[var(--gold)]');
    dots[current].classList.add('bg-[var(--border-gold)]');
    
    current = idx;
    
    cards.forEach(function(c, i) {
      c.classList.remove('exit');
      if (i !== current) c.classList.remove('active');
    });
    cards[current].classList.add('active');
    dots[current].classList.remove('bg-[var(--border-gold)]');
    dots[current].classList.add('bg-[var(--gold)]');
  }
  
  dots.forEach(function(dot, i) { dot.addEventListener('click', function() { goTo(i); }); });
  
  setInterval(function() { goTo((current + 1) % testimonials.length); }, 6000);
}

/* ============================================
   SMOOTH SCROLL
============================================ */
function initSmoothScroll() {
  document.querySelectorAll('a[href^="#"]').forEach(function(a) {
    a.addEventListener('click', function(e) {
      var target = document.querySelector(a.getAttribute('href'));
      if (target) {
        e.preventDefault();
        target.scrollIntoView({ behavior: 'smooth' });
      }
    });
  });
}

/* ============================================
   HIDE LOADER (failsafe)
============================================ */
function hideLoader() {
  var loader = document.getElementById('loader');
  if (loader && !loader.classList.contains('hidden')) {
    loader.classList.add('hidden');
  }
}
